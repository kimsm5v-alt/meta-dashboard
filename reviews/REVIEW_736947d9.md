**문제 1: 메모리 누수 및 경쟁 조건**
```typescript
// 현재 코드 (문제 있음)
useEffect(() => {
  const loadResult = async () => {
    // API 호출들...
  };
  loadResult();
}, [user]);

// 제안하는 수정 코드
useEffect(() => {
  let isMounted = true;
  const abortController = new AbortController();
  
  const loadResult = async () => {
    if (!isMounted) return;
    // API 호출 시 signal 추가
    const groups = await getMyGroups(user.id, { signal: abortController.signal });
  };
  
  loadResult();
  
  return () => {
    isMounted = false;
    abortController.abort();
  };
}, [user]);
```
